115inline void replace(std::string & str, const std::string & from, const std::string & to)
116{
117 if (from.empty()) {return;}
118 size_t start_pos = 0;
119 while ((start_pos = str.find(from, start_pos)) != std::string::npos) {
120 str.replace(start_pos, from.length(), to);
121 start_pos += to.length();
122 }
123}
