138inline void replace(std::string & str, const std::string & from, const std::string & to)
139{
140 if (from.empty()) {return;}
141 size_t start_pos = 0;
142 while ((start_pos = str.find(from, start_pos)) != std::string::npos) {
143 str.replace(start_pos, from.length(), to);
144 start_pos += to.length();
145 }
146}
