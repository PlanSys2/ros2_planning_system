52inline void from_json(const nlohmann::json & js, std::chrono::milliseconds & dest)
53{
54 if (js.contains("ms")) {
55 dest = std::chrono::milliseconds(js.at("ms").get<int>());
56 } else {
57 throw std::runtime_error("Invalid JSON for std::chrono::milliseconds");
58 }
59}

61inline void to_json(nlohmann::json & js, const std::chrono::milliseconds & src)
62{
63 js["__type"] = "std::chrono::milliseconds";
64 js["ms"] = src.count();
65}
