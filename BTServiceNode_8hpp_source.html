90 void initialize()
91 {
92 // Get the required items from the blackboard
93 auto bt_loop_duration =
94 config().blackboard->template get<std::chrono::milliseconds>("bt_loop_duration");
95 getInputOrBlackboard("server_timeout", server_timeout_);
96 wait_for_service_timeout_ =
97 config().blackboard->template get<std::chrono::milliseconds>("wait_for_service_timeout");
98
99 // timeout should be less than bt_loop_duration to be able to finish the current tick
100 max_timeout_ = std::chrono::duration_cast<std::chrono::milliseconds>(bt_loop_duration * 0.5);
101
102 // Now that we have node_ to use, create the service client for this BT service
103 createROSInterfaces();
104 }

109 void createROSInterfaces()
110 {
111 std::string service_new;
112 getInput("service_name", service_new);
113 if (service_new != service_name_ || !service_client_) {
114 service_name_ = service_new;
115 node_ = config().blackboard->template get<rclcpp_lifecycle::LifecycleNode::SharedPtr>("node");
116
117 callback_group_ = node_->create_callback_group(
118 rclcpp::CallbackGroupType::MutuallyExclusive, false);
119
120 callback_group_executor_ = std::make_shared<rclcpp::executors::SingleThreadedExecutor>();
121 callback_group_executor_->add_callback_group(
122 callback_group_, node_->get_node_base_interface());
123 service_client_ = node_->create_client<ServiceT>(
124 service_name_, rclcpp::ServicesQoS(), callback_group_);
125 }

158 BT::NodeStatus tick() override
159 {
160 if (!BT::isStatusActive(status())) {
161 initialize();
162 }
163
164 if (!request_sent_) {
165 // reset the flag to send the request or not,
166 // allowing the user the option to set it in on_tick
167 should_send_request_ = true;
168
169 // Clear the input request to make sure we have no leftover from previous calls
170 request_ = std::make_shared<typename ServiceT::Request>();
171
172 // user defined callback, may modify "should_send_request_".
173 on_tick();
174
175 if (!should_send_request_) {
176 return BT::NodeStatus::FAILURE;
177 }
178
179 future_result_ = service_client_->async_send_request(request_).share();
180 sent_time_ = node_->now();
181 request_sent_ = true;
182 }
183 return check_future();
184 }
185
189 void halt() override
190 {
191 request_sent_ = false;
192 resetStatus();
193 }

209 virtual BT::NodeStatus on_completion(std::shared_ptr<typename ServiceT::Response>/*response*/)

218 virtual BT::NodeStatus check_future()
219 {
220 auto elapsed = (node_->now() - sent_time_).template to_chrono<std::chrono::milliseconds>();
221 auto remaining = server_timeout_ - elapsed;
222
223 if (remaining > std::chrono::milliseconds(0)) {
224 auto timeout = remaining > max_timeout_ ? max_timeout_ : remaining;
225
226 auto rc = callback_group_executor_->spin_until_future_complete(future_result_, timeout);
227 if (rc == rclcpp::FutureReturnCode::SUCCESS) {
228 request_sent_ = false;
229 BT::NodeStatus status = on_completion(future_result_.get());
230 return status;
231 }
232
233 if (rc == rclcpp::FutureReturnCode::TIMEOUT) {
234 on_wait_for_result();
235 elapsed = (node_->now() - sent_time_).template to_chrono<std::chrono::milliseconds>();
236 if (elapsed < server_timeout_) {
237 return BT::NodeStatus::RUNNING;
238 }
239 }
240 }
241
242 RCLCPP_WARN(
243 node_->get_logger(),
244 "Node timed out while executing service call to %s.", service_name_.c_str());
245 request_sent_ = false;
246 return BT::NodeStatus::FAILURE;
247 }
