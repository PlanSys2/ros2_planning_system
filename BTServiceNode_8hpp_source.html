87 void initialize()
88 {
89 // Get the required items from the blackboard
90 auto bt_loop_duration =
91 config().blackboard->template get<std::chrono::milliseconds>("bt_loop_duration");
92 getInputOrBlackboard("server_timeout", server_timeout_);
93 wait_for_service_timeout_ =
94 config().blackboard->template get<std::chrono::milliseconds>("wait_for_service_timeout");
95
96 // timeout should be less than bt_loop_duration to be able to finish the current tick
97 max_timeout_ = std::chrono::duration_cast<std::chrono::milliseconds>(bt_loop_duration * 0.5);

99 // Now that we have node_ to use, create the service client for this BT service
100 createROSInterfaces();

106 void createROSInterfaces()
107 {
108 // The service_name port, if given, overrides the name passed to the constructor
109 std::string service_new;
110 if (!getInput("service_name", service_new) || service_new.empty()) {
111 service_new = service_name_;
112 }
113 if (service_new != service_name_ || !service_client_) {
114 service_name_ = service_new;
115 if (!config().blackboard->get("node", node_) || !node_) {
116 throw BT::RuntimeError(service_node_name_ + ": no 'node' in the blackboard");
117 }
118
119 callback_group_ = node_->create_callback_group(
120 rclcpp::CallbackGroupType::MutuallyExclusive, false);
121
122 callback_group_executor_ = std::make_shared<rclcpp::executors::SingleThreadedExecutor>();
123 callback_group_executor_->add_callback_group(
124 callback_group_, node_->get_node_base_interface());
125 service_client_ = node_->create_client<ServiceT>(
126 service_name_, rclcpp::ServicesQoS(), callback_group_);
127 }

160 BT::NodeStatus tick() override
161 {
162 if (!BT::isStatusActive(status())) {
163 initialize();
164 }
165
166 if (!request_sent_) {
167 // reset the flag to send the request or not,
168 // allowing the user the option to set it in on_tick
169 should_send_request_ = true;
170
171 // Clear the input request to make sure we have no leftover from previous calls
172 request_ = std::make_shared<typename ServiceT::Request>();
173
174 // user defined callback, may modify "should_send_request_".
175 on_tick();
176
177 if (!should_send_request_) {
178 return BT::NodeStatus::FAILURE;
179 }
180
181 future_result_ = service_client_->async_send_request(request_).share();
182 sent_time_ = node_->now();
183 request_sent_ = true;
184 }
185 return check_future();
186 }
187
191 void halt() override
192 {
193 if (request_sent_) {
194 service_client_->prune_pending_requests();
195 }
196 request_sent_ = false;
197 resetStatus();
198 }

223 virtual BT::NodeStatus check_future()
224 {
225 auto elapsed = (node_->now() - sent_time_).template to_chrono<std::chrono::milliseconds>();
226 auto remaining = server_timeout_ - elapsed;
227
228 if (remaining > std::chrono::milliseconds(0)) {
229 auto timeout = remaining > max_timeout_ ? max_timeout_ : remaining;
230
231 auto rc = callback_group_executor_->spin_until_future_complete(future_result_, timeout);
232 if (rc == rclcpp::FutureReturnCode::SUCCESS) {
233 request_sent_ = false;
234 BT::NodeStatus status = on_completion(future_result_.get());
235 return status;
236 }
237
238 if (rc == rclcpp::FutureReturnCode::TIMEOUT) {
239 on_wait_for_result();
240 elapsed = (node_->now() - sent_time_).template to_chrono<std::chrono::milliseconds>();
241 if (elapsed < server_timeout_) {
242 return BT::NodeStatus::RUNNING;
243 }
244 }
245 }
246
247 RCLCPP_WARN(
248 node_->get_logger(),
249 "Node timed out while executing service call to %s.", service_name_.c_str());
250 // A late response would stay pending for ever
251 service_client_->prune_pending_requests();
252 request_sent_ = false;
253 return BT::NodeStatus::FAILURE;
254 }
