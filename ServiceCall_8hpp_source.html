63// What a component answers (error_info) to a request it does not take because it is not active
64constexpr char kNotActive[] = "Requesting service in non-active state";
65
66// Whether a response says that its component is not active (responses without error_info never)
67template<typename ResponseT, typename = void>
68struct HasErrorInfo : std::false_type {};
69template<typename ResponseT>
70struct HasErrorInfo<ResponseT, std::void_t<decltype(std::declval<ResponseT>().error_info)>>
71 : std::true_type {};
72
73template<typename ResponseT>
74bool says_not_active(const ResponseT & response)
75{
76 if constexpr (HasErrorInfo<ResponseT>::value) {
77 return response.error_info == kNotActive;
78 } else {
79 return false;
80 }
81}
82
83// Whether a response has a success field
84template<typename ResponseT, typename = void>
85struct HasSuccess : std::false_type {};
86template<typename ResponseT>
87struct HasSuccess<ResponseT, std::void_t<decltype(std::declval<ResponseT>().success)>>
88 : std::true_type {};
89
90// A response that says the request failed, and why when it has error_info
91template<typename ResponseT>
92ResponseT failed_response(const std::string & error)
93{
94 ResponseT response;
95 if constexpr (HasSuccess<ResponseT>::value) {
96 response.success = false;
97 }
98 if constexpr (HasErrorInfo<ResponseT>::value) {
99 response.error_info = error;
100 }
101 return response;
102}

109template<typename ServiceT, typename CallbackT>
110auto guarded(const rclcpp::Logger & logger, CallbackT callback)
111{
112 return [logger, callback = std::move(callback)](
113 const std::shared_ptr<rmw_request_id_t> header,
114 const std::shared_ptr<typename ServiceT::Request> request,
115 std::shared_ptr<typename ServiceT::Response> response) {
116 try {
117 callback(header, request, response);
118 } catch (const std::exception & e) {
119 RCLCPP_ERROR(logger, "Request rejected: %s", e.what());
120 *response = failed_response<typename ServiceT::Response>(e.what());
121 }
122 };
123}
124
128template<typename ServiceT, typename CallbackT>
129auto guarded_deferred(const rclcpp::Logger & logger, CallbackT callback)
130{
131 return [logger, callback = std::move(callback)](
132 const std::shared_ptr<rclcpp::Service<ServiceT>> service,
133 const std::shared_ptr<rmw_request_id_t> header,
134 const std::shared_ptr<typename ServiceT::Request> request) {
135 try {
136 callback(service, header, request);
137 } catch (const std::exception & e) {
138 RCLCPP_ERROR(logger, "Request rejected: %s", e.what());
139 auto response = failed_response<typename ServiceT::Response>(e.what());
140 service->send_response(*header, response);
141 }
142 };

152class ActiveComponent
153{
154public:
155 ActiveComponent(std::shared_ptr<ClientNode> node, const std::string & component)
156 : node_(node), component_(component),
157 client_(node->create_client<lifecycle_msgs::srv::GetState>(component + "/get_state"))
158 {
159 }
160
165 bool wait()
166 {
167 if (active_) {
168 return true;
169 }
170 // One thread asks; the others find it active when they get the lock
171 std::lock_guard<std::mutex> lock(mutex_);
172 const auto since = std::chrono::steady_clock::now();
173 auto last_log = since;
174 while (!active_ && node_->running()) {
175 // Given up once: the calls after ask once, and do not wait again, until it is back
176 const auto limit = gave_up_ ? std::chrono::milliseconds(0) : max_wait_;
177 if (max_wait_.count() > 0 && std::chrono::steady_clock::now() - since >= limit) {
178 if (!gave_up_) {
179 RCLCPP_ERROR(
180 node_->get_logger(), "%s is not active after %.0f s: giving up", component_.c_str(),
181 std::chrono::duration<double>(max_wait_).count());
182 }
183 gave_up_ = true;
184 // Asked at most once a second meanwhile: each call fails at once
185 const auto now = std::chrono::steady_clock::now();
186 if (now - last_ask_ < std::chrono::seconds(1)) {
187 return false;
188 }
189 last_ask_ = now;
190 if (!ask()) {
191 return false;
192 }
193 active_ = true;
194 gave_up_ = false;
195 break;
196 }
197 // At most one question every kPeriod, also when a wait ends at once (shutdown, no node)
198 const auto next = std::chrono::steady_clock::now() + kPeriod;
199 last_ask_ = std::chrono::steady_clock::now();
200 if (ask()) {
201 active_ = true;
202 gave_up_ = false;
203 break;
204 }
205 std::this_thread::sleep_until(next);
206 if (std::chrono::steady_clock::now() - last_log > kLogPeriod) {
207 RCLCPP_INFO(node_->get_logger(), "Waiting for %s to be active...", component_.c_str());
208 last_log = std::chrono::steady_clock::now();
209 }
210 }
211 return active_;
212 }
213
217 void lost() {active_ = false;}
218
222 void set_max_wait(std::chrono::milliseconds max_wait) {max_wait_ = max_wait;}
223
224 const std::string & component() const {return component_;}
225
226private:
227 // Whether it answers that it is active
228 bool ask()
229 {
230 if (!wait_for_server(client_, std::chrono::seconds(1))) {
231 return false;
232 }
233 try {
234 auto future = client_->async_send_request(
235 std::make_shared<lifecycle_msgs::srv::GetState::Request>());
236 if (node_->wait(future, std::chrono::seconds(1)) == rclcpp::FutureReturnCode::SUCCESS) {
237 return future.get()->current_state.id == lifecycle_msgs::msg::State::PRIMARY_STATE_ACTIVE;
238 }
239 // A lost answer: a new client, discovered again (see ServiceClient)
240 client_->remove_pending_request(future);
241 client_ = node_->create_client<lifecycle_msgs::srv::GetState>(component_ + "/get_state");
242 } catch (const std::runtime_error &) {
243 // ROS shutting down, or the node gone
244 }
245 return false;
246 }
247
248 static constexpr std::chrono::milliseconds kPeriod {500};
249 static constexpr std::chrono::seconds kLogPeriod {5};
250 std::shared_ptr<ClientNode> node_;
251 std::string component_;
252 std::chrono::milliseconds max_wait_ {0};
253 bool gave_up_ {false};
254 std::chrono::steady_clock::time_point last_ask_;
255 std::atomic<bool> active_ {false};
256 std::mutex mutex_;
257 rclcpp::Client<lifecycle_msgs::srv::GetState>::SharedPtr client_;
258};

274template<typename ServiceT>
275class ServiceClient
276{
277public:
278 using SharedPtr = std::shared_ptr<ServiceClient<ServiceT>>;
279 using Response = typename ServiceT::Response::SharedPtr;
280 using Request = typename ServiceT::Request::SharedPtr;
281
282 ServiceClient(
283 std::shared_ptr<ClientNode> node, const std::string & name,
284 std::shared_ptr<ActiveComponent> active = nullptr)
285 : node_(node), name_(name), active_(active), client_(node->create_client<ServiceT>(name))
286 {
287 }
288
299 Response call(
300 const Request & request,
301 std::chrono::nanoseconds response_timeout = std::chrono::seconds(1),
302 int attempts = 3,
303 std::chrono::nanoseconds service_timeout = std::chrono::seconds(5))
304 {
305 bool waited_again = false;
306 int not_active_waits = 0;
307 for (int attempt = 1; attempt <= attempts && node_->running(); attempt++) {
308 if (active_ && !active_->wait()) {
309 return nullptr;
310 }
311 auto client = current();
312 if (!wait_for_server(client, service_timeout)) {
313 if (active_ && !waited_again && node_->running()) {
314 // Gone (PlanSys2 stopped or restarted): back to waiting for it, once; active and still
315 // without this service, it is an error
316 active_->lost();
317 waited_again = true;
318 attempt--;
319 continue;
320 }
321 if (node_->running()) {
322 RCLCPP_ERROR_STREAM(node_->get_logger(), name() << ": service not available");
323 }
324 return nullptr;
325 }
326 try {
327 auto future = client->async_send_request(request);
328 if (node_->wait(future, response_timeout) == rclcpp::FutureReturnCode::SUCCESS) {
329 auto response = future.get();
330 if (active_ && says_not_active(*response) && not_active_waits++ < kNotActiveWaits) {
331 // Deactivated since it was seen active: wait for it again, and ask again
332 active_->lost();
333 attempt--;
334 continue;
335 }
336 return response;
337 }
338 // Otherwise the late answer would stay pending forever
339 client->remove_pending_request(future);
340 } catch (const std::exception & e) {
341 // rclcpp shutting down throws from the graph: no answer, not the end of the process
342 if (node_->running()) {
343 RCLCPP_ERROR_STREAM(node_->get_logger(), name() << ": " << e.what());
344 }
345 return nullptr;
346 }
347 if (!node_->running()) {
348 return nullptr;
349 }
350 if (!renew(client)) {
351 return nullptr;
352 }
353 if (attempt < attempts) {
354 RCLCPP_WARN_STREAM(
355 node_->get_logger(), name() << ": no response, sending it again with a new client (" <<
356 attempt + 1 << "/" << attempts << ")");
357 }
358 }
359 if (active_) {
360 active_->lost(); // The next call checks that it is still active
361 }
362 if (node_->running()) {
363 RCLCPP_ERROR_STREAM(node_->get_logger(), name() << ": no response");
364 }
365 return nullptr;
366 }
367
371 bool wait_for_service(std::chrono::nanoseconds timeout)
372 {
373 return wait_for_server(current(), timeout);
374 }
375
376 bool service_is_ready() const {return current()->service_is_ready();}
377
378 std::string name() const {return current()->get_service_name();}
379
380private:
381 typename rclcpp::Client<ServiceT>::SharedPtr current() const
382 {
383 std::lock_guard<std::mutex> lock(mutex_);
384 return client_;
385 }
386
387 // A new client, unless another call already made one after this one failed
388 bool renew(const typename rclcpp::Client<ServiceT>::SharedPtr & failed)
389 {
390 std::lock_guard<std::mutex> lock(mutex_);
391 if (client_ != failed) {
392 return true;
393 }
394 try {
395 client_ = node_->create_client<ServiceT>(name_);
396 } catch (const std::runtime_error &) {
397 return false; // The node is gone
398 }
399 return true;
400 }
401
402 // A component that says it is active and answers that it is not, so many times: an error
403 static constexpr int kNotActiveWaits = 3;
404 std::shared_ptr<ClientNode> node_;
405 std::string name_;
406 std::shared_ptr<ActiveComponent> active_;
407 mutable std::mutex mutex_;
408 typename rclcpp::Client<ServiceT>::SharedPtr client_;
409};
