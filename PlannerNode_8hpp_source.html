228template<typename NodeT>
229std::string get_plugin_type_param(
230 NodeT node,
231 const std::string & plugin_name)
232{
233 declare_parameter_if_not_declared(node, plugin_name + ".plugin", rclcpp::ParameterValue(""));
234 std::string plugin_type;
235 if (!node->get_parameter(plugin_name + ".plugin", plugin_type)) {
236 RCLCPP_FATAL(node->get_logger(), "'plugin' param not defined for %s", plugin_name.c_str());
237 exit(-1);
238 }
239 return plugin_type;
240}
