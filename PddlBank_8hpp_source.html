64inline std::vector<BankCase> read_pddl_bank(const std::filesystem::path & root)
65{
66 std::vector<BankCase> cases;
67 std::error_code error;
68 for (const auto & entry : std::filesystem::directory_iterator(root, error)) {
69 const auto list = entry.path() / "cases.tsv";
70 if (!std::filesystem::exists(list)) {
71 continue;
72 }
73 std::ifstream in(list);
74 std::string line;
75 while (std::getline(in, line)) {
76 std::vector<std::string> fields;
77 std::stringstream row(line);
78 std::string field;
79 while (std::getline(row, field, '\t')) {
80 fields.push_back(field);
81 }
82 if (fields.size() < 4) {
83 continue;
84 }
85 BankCase task;
86 task.feature = entry.path().filename().string();
87 task.name = task.feature + "/" + std::filesystem::path(fields[1]).stem().string();
88 task.domain_file = entry.path() / fields[0];
89 task.problem_file = entry.path() / fields[1];
90 task.domain = read_bank_file(task.domain_file);
91 task.problem = read_bank_file(task.problem_file);
92 task.solvable = fields[2] == "yes";
93 std::stringstream uses(fields[3]);
94 for (std::string requirement; uses >> requirement; ) {
95 if (requirement != "-") {
96 task.uses.insert(requirement);
97 }
98 }
99 task.description = fields.size() > 4 ? fields[4] : "";
100 cases.push_back(task);
101 }
102 }
103 std::sort(
104 cases.begin(), cases.end(),
105 [](const BankCase & a, const BankCase & b) {return a.name < b.name;});
106 return cases;
107}
