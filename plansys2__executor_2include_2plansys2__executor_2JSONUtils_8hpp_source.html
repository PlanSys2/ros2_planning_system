53inline void from_json(const nlohmann::json & js, std::chrono::milliseconds & dest)
54{
55 if (js.contains("ms")) {
56 dest = std::chrono::milliseconds(js.at("ms").get<int>());
57 } else {
58 throw std::runtime_error("Invalid JSON for std::chrono::milliseconds");
59 }
60}
61
62inline void to_json(nlohmann::json & js, const std::chrono::milliseconds & src)
63{
64 js["__type"] = "std::chrono::milliseconds";
65 js["ms"] = src.count();
66}

68inline void from_json(
69 const nlohmann::json & js,
70 std::unordered_map<std::string, std::string> & dest)
71{
72 if (js.is_object()) {
73 dest.clear();
74 for (auto & [key, value] : js.items()) {
75 if (value.is_string()) {
76 dest[key] = value.get<std::string>();
77 } else {
78 throw std::runtime_error(
79 "Invalid JSON for std::unordered_map<std::string, std::string>: value must be string");
80 }
81 }
82 } else {
83 throw std::runtime_error(
84 "Invalid JSON for std::unordered_map<std::string, std::string>: must be object");
85 }
86}
87
88inline void to_json(nlohmann::json & js, const std::unordered_map<std::string, std::string> & src)
89{
90 js = nlohmann::json::object();
91 for (const auto & [key, value] : src) {
92 js[key] = value;
93 }
94}
