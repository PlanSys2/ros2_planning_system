416struct hash<plansys2::State>
417{
418 std::size_t operator()(const plansys2::State & state) const
419 {
420 std::size_t seed = 0;
421 seed ^= unordered_container_hash(state.instances_);
422 seed ^= unordered_container_hash(state.functions_);
423 seed ^= unordered_container_hash(state.predicates_);
424 seed ^= unordered_container_hash(state.inferred_predicates_);
425 seed ^= unordered_container_hash(state.union_predicates_inferred_predicates_);
426 seed ^= std::hash<decltype(state.derived_predicates_)>{}(state.derived_predicates_);
427 return seed;
428 }
