34template<typename T>
35inline void hash_combine(std::size_t & seed, const T & value)
36{
37 seed ^= std::hash<T>{}(value) + 0x9e3779b9 + (seed << 6) + (seed >> 2);
38}
39
40inline std::size_t hash_node(const plansys2_msgs::msg::Node & node)
41{
42 std::size_t seed = 0;
43
44 hash_combine(seed, node.name);
45 hash_combine(seed, node.node_type);
46 hash_combine(seed, node.children.size());
47 hash_combine(seed, node.parameters.size());
48
49 for (const auto & param : node.parameters) {
50 hash_combine(seed, param.name);
51 }
52 return seed;
53}

55template<typename UnorderedContainer>
56inline std::size_t unordered_container_hash(const UnorderedContainer & container)
57{
58 std::size_t h = 0;
59 for (const auto & elem : container) {
60 // Combine using XOR for order independence
61 h ^= std::hash<std::decay_t<decltype(elem)>>{}(elem);
62 }
63 // Optionally, mix in the size for more uniqueness
64 h ^= container.size();
65 return h;
66}
67
68// For unordered_map and similar
69template<typename Key, typename T, typename Hash, typename KeyEqual, typename Alloc>
70std::size_t unordered_container_hash(const std::unordered_map<Key, T, Hash, KeyEqual, Alloc> & map)
71{
72 std::size_t h = 0;
73 for (const auto & [key, value] : map) {
74 std::size_t entry_hash = std::hash<Key>{}(key);
75 entry_hash ^= std::hash<T>{}(value) + 0x9e3779b9;
76 h ^= entry_hash;
77 }
78 h ^= map.size();
79 return h;
80}

205class Derived : public plansys2_msgs::msg::Derived
206{
207public:
208 Derived()
209 : plansys2_msgs::msg::Derived() {computeNormalizedDerived();}
210 Derived(const plansys2_msgs::msg::Derived & derived) // NOLINT(runtime/explicit)
211 : plansys2_msgs::msg::Derived(derived)
212 {
213 computeNormalizedDerived();
214 }
215
216 bool operator==(const Derived & d) const
217 {
218 if (this == &d) {return true;}
219 if (!normalizedDerivedComputed()) {
220 computeNormalizedDerived();
221 }
222 if (!d.normalizedDerivedComputed()) {
223 d.computeNormalizedDerived();
224 }
225 return parser::pddl::checkNodeEquality(getNormalizedPredicate(), d.getNormalizedPredicate()) &&
226 parser::pddl::checkTreeEquality(
227 getNormalizedPreconditions(), d.getNormalizedPreconditions());
228 }
229
230 const size_t & getNormalizedHash() const {return normalized_hash_;}
231
232 const plansys2_msgs::msg::Node & getNormalizedPredicate() const {return normalized_predicate_;}
233
234 const plansys2_msgs::msg::Tree & getNormalizedPreconditions() const
235 {
236 return normalized_preconditions_;
237 }
238
239 Derived getNormalizedDerived()
240 {
241 if (!normalizedDerivedComputed()) {
242 computeNormalizedDerived();
243 }
244 Derived new_derived;
245 new_derived.predicate = normalized_predicate_;
246 new_derived.preconditions = normalized_preconditions_;
247 return new_derived;
248 }

250 bool normalizedDerivedComputed() const
251 {
252 return normalized_predicate_.name != "" && !normalized_preconditions_.nodes.empty();
253 }
254
255 std::string normalize_param(
256 const std::string & old_name, std::map<std::string, std::string> & var_map, uint & i) const
257 {
258 if (old_name.empty() || old_name.front() != '?') {return old_name;}
259
260 auto it = var_map.find(old_name);
261 if (it != var_map.end()) {return it->second;}
262
263 std::string normalized = "?" + std::to_string(i++);
264 var_map[old_name] = normalized;
265 return normalized;
266 }
267
268 void computeNormalizedDerived() const
269 {
270 if (normalizedDerivedComputed()) {return;}
271
272 normalized_predicate_ = this->predicate;
273 normalized_preconditions_ = this->preconditions;
274
275 std::map<std::string, std::string> var_map;
276 uint i = 0;
277
278 // Normalize predicate parameters
279 for (auto & p : normalized_predicate_.parameters) {
280 p.name = this->normalize_param(p.name, var_map, i);
281 }
282
283 // Normalize precondition nodes and their parameters
284 for (auto & node : normalized_preconditions_.nodes) {
285 if (node.node_type == plansys2_msgs::msg::Node::PARAMETER) {
286 node.name = this->normalize_param(node.name, var_map, i);
287 }
288
289 for (auto & p : node.parameters) {p.name = this->normalize_param(p.name, var_map, i);}
290 }
291
292 normalized_hash_ = 0;
293 std::hash_combine(normalized_hash_, std::hash_node(normalized_predicate_));
294 std::hash_combine(normalized_hash_, normalized_preconditions_.nodes.size());
295 for (auto & node : normalized_preconditions_.nodes) {
296 std::hash_combine(normalized_hash_, std::hash_node(node));
297 }
298 }
299
300 std::string toNormalizedString() const
301 {
302 if (!normalizedDerivedComputed()) {
303 computeNormalizedDerived();
304 }
305
306 std::ostringstream oss;
307
308 // Predicate
309 oss << "(" << normalized_predicate_.name;
310 for (const auto & p : normalized_predicate_.parameters) {
311 oss << " " << p.name;
312 }
313 oss << ")";
314
315 // Preconditions
316 if (!normalized_preconditions_.nodes.empty()) {
317 oss << " :- ";
318
319 bool first = true;
320 for (const auto & node : normalized_preconditions_.nodes) {
321 if (!first) {
322 oss << " ";
323 }
324 first = false;
325
326 oss << nodeToString(node);
327 }
328 }
329
330 return oss.str();
331 }
332
333private:
334 mutable plansys2_msgs::msg::Node normalized_predicate_;
335 mutable plansys2_msgs::msg::Tree normalized_preconditions_;
336 mutable bool normalized_cache_;
337 mutable std::size_t normalized_hash_;
338
339 static std::string
340 nodeToString(const plansys2_msgs::msg::Node & node)
341 {
342 std::ostringstream oss;
343
344 switch (node.node_type) {
345 case plansys2_msgs::msg::Node::AND:
346 oss << "(and";
347 for (const auto & p : node.parameters) {
348 oss << " " << p.name;
349 }
350 oss << ")";
351 break;
352
353 case plansys2_msgs::msg::Node::OR:
354 oss << "(or";
355 for (const auto & p : node.parameters) {
356 oss << " " << p.name;
357 }
358 oss << ")";
359 break;
360
361 case plansys2_msgs::msg::Node::PREDICATE:
362 oss << "(" << node.name;
363 for (const auto & p : node.parameters) {
364 oss << " " << p.name;
365 }
366 oss << ")";
367 break;
368
369 case plansys2_msgs::msg::Node::PARAMETER:
370 oss << node.name;
371 break;
372
373 default:
374 oss << "(unknown)";
375 break;
376 }
377
378 return oss.str();
379 }
