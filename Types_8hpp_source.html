36template<typename T>
37inline void hash_combine(std::size_t & seed, const T & value)
38{
39 seed ^= std::hash<T>{}(value) + 0x9e3779b9 + (seed << 6) + (seed >> 2);
40}
41
42inline std::size_t hash_node(const plansys2_msgs::msg::Node & node)
43{
44 std::size_t seed = 0;
45
46 plansys2::hash_combine(seed, node.name);
47 plansys2::hash_combine(seed, node.node_type);
48 plansys2::hash_combine(seed, node.children.size());
49 plansys2::hash_combine(seed, node.parameters.size());
50
51 for (const auto & param : node.parameters) {
52 plansys2::hash_combine(seed, param.name);
53 }
54 return seed;
55}

57template<typename UnorderedContainer>
58inline std::size_t unordered_container_hash(const UnorderedContainer & container)
59{
60 std::size_t h = 0;
61 for (const auto & elem : container) {
62 // Combine using XOR for order independence
63 h ^= std::hash<std::decay_t<decltype(elem)>>{}(elem);
64 }
65 // Optionally, mix in the size for more uniqueness
66 h ^= container.size();
67 return h;
68}
69
70// For unordered_map and similar
71template<typename Key, typename T, typename Hash, typename KeyEqual, typename Alloc>
72std::size_t unordered_container_hash(const std::unordered_map<Key, T, Hash, KeyEqual, Alloc> & map)
73{
74 std::size_t h = 0;
75 for (const auto & [key, value] : map) {
76 std::size_t entry_hash = std::hash<Key>{}(key);
77 entry_hash ^= std::hash<T>{}(value) + 0x9e3779b9;
78 h ^= entry_hash;
79 }
80 h ^= map.size();
81 return h;
82}

207class Derived : public plansys2_msgs::msg::Derived
208{
209public:
210 Derived()
211 : plansys2_msgs::msg::Derived() {computeNormalizedDerived();}
212 Derived(const plansys2_msgs::msg::Derived & derived) // NOLINT(runtime/explicit)
213 : plansys2_msgs::msg::Derived(derived)
214 {
215 computeNormalizedDerived();
216 }
217
218 bool operator==(const Derived & d) const
219 {
220 if (this == &d) {return true;}
221 if (!normalizedDerivedComputed()) {
222 computeNormalizedDerived();
223 }
224 if (!d.normalizedDerivedComputed()) {
225 d.computeNormalizedDerived();
226 }
227 return parser::pddl::checkNodeEquality(getNormalizedPredicate(), d.getNormalizedPredicate()) &&
228 parser::pddl::checkTreeEquality(
229 getNormalizedPreconditions(), d.getNormalizedPreconditions());
230 }
231
232 const size_t & getNormalizedHash() const {return normalized_hash_;}
233
234 const plansys2_msgs::msg::Node & getNormalizedPredicate() const {return normalized_predicate_;}
235
236 const plansys2_msgs::msg::Tree & getNormalizedPreconditions() const
237 {
238 return normalized_preconditions_;
239 }
240
241 Derived getNormalizedDerived()
242 {
243 if (!normalizedDerivedComputed()) {
244 computeNormalizedDerived();
245 }
246 Derived new_derived;
247 new_derived.predicate = normalized_predicate_;
248 new_derived.preconditions = normalized_preconditions_;
249 return new_derived;
250 }
251
252 bool normalizedDerivedComputed() const
253 {
254 return normalized_predicate_.name != "" && !normalized_preconditions_.nodes.empty();
255 }
256
257 std::string normalize_param(
258 const std::string & old_name, std::map<std::string, std::string> & var_map, uint & i) const
259 {
260 if (old_name.empty() || old_name.front() != '?') {return old_name;}
261
262 auto it = var_map.find(old_name);
263 if (it != var_map.end()) {return it->second;}
264
265 std::string normalized = "?" + std::to_string(i++);
266 var_map[old_name] = normalized;
267 return normalized;
268 }
269
270 void computeNormalizedDerived() const
271 {
272 if (normalizedDerivedComputed()) {return;}
273
274 normalized_predicate_ = this->predicate;
275 normalized_preconditions_ = this->preconditions;
276
277 std::map<std::string, std::string> var_map;
278 uint i = 0;
279
280 // Normalize predicate parameters
281 for (auto & p : normalized_predicate_.parameters) {
282 p.name = this->normalize_param(p.name, var_map, i);
283 }
284
285 // Normalize precondition nodes and their parameters
286 for (auto & node : normalized_preconditions_.nodes) {
287 if (node.node_type == plansys2_msgs::msg::Node::PARAMETER) {
288 node.name = this->normalize_param(node.name, var_map, i);
289 }
290
291 for (auto & p : node.parameters) {p.name = this->normalize_param(p.name, var_map, i);}
292 }
293
294 normalized_hash_ = 0;
295 plansys2::hash_combine(normalized_hash_, plansys2::hash_node(normalized_predicate_));
296 plansys2::hash_combine(normalized_hash_, normalized_preconditions_.nodes.size());
297 for (auto & node : normalized_preconditions_.nodes) {
298 plansys2::hash_combine(normalized_hash_, plansys2::hash_node(node));
299 }
300 }
301
302 std::string toNormalizedString() const
303 {
304 if (!normalizedDerivedComputed()) {
305 computeNormalizedDerived();
306 }
307
308 std::ostringstream oss;
309
310 // Predicate
311 oss << "(" << normalized_predicate_.name;
312 for (const auto & p : normalized_predicate_.parameters) {
313 oss << " " << p.name;
314 }
315 oss << ")";
316
317 // Preconditions
318 if (!normalized_preconditions_.nodes.empty()) {
319 oss << " :- ";
320
321 bool first = true;
322 for (const auto & node : normalized_preconditions_.nodes) {
323 if (!first) {
324 oss << " ";
325 }
326 first = false;
327
328 oss << nodeToString(node);
329 }
330 }
331
332 return oss.str();
333 }
334
335private:
336 mutable plansys2_msgs::msg::Node normalized_predicate_;
337 mutable plansys2_msgs::msg::Tree normalized_preconditions_;
338 mutable std::size_t normalized_hash_ {0};
339
340 static std::string
341 nodeToString(const plansys2_msgs::msg::Node & node)
342 {
343 std::ostringstream oss;
344
345 switch (node.node_type) {
346 case plansys2_msgs::msg::Node::AND:
347 oss << "(and";
348 for (const auto & p : node.parameters) {
349 oss << " " << p.name;
350 }
351 oss << ")";
352 break;
353
354 case plansys2_msgs::msg::Node::OR:
355 oss << "(or";
356 for (const auto & p : node.parameters) {
357 oss << " " << p.name;
358 }
359 oss << ")";
360 break;
361
362 case plansys2_msgs::msg::Node::PREDICATE:
363 oss << "(" << node.name;
364 for (const auto & p : node.parameters) {
365 oss << " " << p.name;
366 }
367 oss << ")";
368 break;
369
370 case plansys2_msgs::msg::Node::PARAMETER:
371 oss << node.name;
372 break;
373
374 default:
375 oss << "(unknown)";
376 break;
377 }
378
379 return oss.str();
380 }
