33template<typename T>
34inline void hash_combine(std::size_t & seed, const T & value)
35{
36 seed ^= std::hash<T>{}(value) + 0x9e3779b9 + (seed << 6) + (seed >> 2);
37}
38
39inline std::size_t hash_node(const plansys2_msgs::msg::Node & node)
40{
41 std::size_t seed = 0;
42
43 hash_combine(seed, node.name);
44 hash_combine(seed, node.node_type);
45 hash_combine(seed, node.children.size());
46 hash_combine(seed, node.parameters.size());
47
48 for (const auto & param : node.parameters) {
49 hash_combine(seed, param.name);
50 }
51 return seed;
52}

54template<typename UnorderedContainer>
55inline std::size_t unordered_container_hash(const UnorderedContainer & container)
56{
57 std::size_t h = 0;
58 for (const auto & elem : container) {
59 // Combine using XOR for order independence
60 h ^= std::hash<std::decay_t<decltype(elem)>>{}(elem);
61 }
62 // Optionally, mix in the size for more uniqueness
63 h ^= container.size();
64 return h;
65}
66
67// For unordered_map and similar
68template<typename Key, typename T, typename Hash, typename KeyEqual, typename Alloc>
69std::size_t unordered_container_hash(const std::unordered_map<Key, T, Hash, KeyEqual, Alloc> & map)
70{
71 std::size_t h = 0;
72 for (const auto & [key, value] : map) {
73 std::size_t entry_hash = std::hash<Key>{}(key);
74 entry_hash ^= std::hash<T>{}(value) + 0x9e3779b9;
75 h ^= entry_hash;
76 }
77 h ^= map.size();
78 return h;
79}

204class Derived : public plansys2_msgs::msg::Derived
205{
206public:
207 Derived()
208 : plansys2_msgs::msg::Derived() {computeNormalizedDerived();}
209 Derived(const plansys2_msgs::msg::Derived & derived) // NOLINT(runtime/explicit)
210 : plansys2_msgs::msg::Derived(derived)
211 {
212 computeNormalizedDerived();
213 }
214
215 bool operator==(const Derived & d) const
216 {
217 if (this == &d) {return true;}
218 if (!normalizedDerivedComputed()) {
219 computeNormalizedDerived();
220 }
221 if (!d.normalizedDerivedComputed()) {
222 d.computeNormalizedDerived();
223 }
224 return parser::pddl::checkNodeEquality(getNormalizedPredicate(), d.getNormalizedPredicate()) &&
225 parser::pddl::checkTreeEquality(
226 getNormalizedPreconditions(), d.getNormalizedPreconditions());
227 }
228
229 const size_t & getNormalizedHash() const {return normalized_hash_;}
230
231 const plansys2_msgs::msg::Node & getNormalizedPredicate() const {return normalized_predicate_;}
232
233 const plansys2_msgs::msg::Tree & getNormalizedPreconditions() const
234 {
235 return normalized_preconditions_;
236 }
237
238 Derived getNormalizedDerived()
239 {
240 if (!normalizedDerivedComputed()) {
241 computeNormalizedDerived();
242 }
243 Derived new_derived;
244 new_derived.predicate = normalized_predicate_;
245 new_derived.preconditions = normalized_preconditions_;
246 return new_derived;
247 }

249 bool normalizedDerivedComputed() const
250 {
251 return normalized_predicate_.name != "" && !normalized_preconditions_.nodes.empty();
252 }
253
254 std::string normalize_param(
255 const std::string & old_name, std::map<std::string, std::string> & var_map, uint & i) const
256 {
257 if (old_name.empty() || old_name.front() != '?') {return old_name;}
258
259 auto it = var_map.find(old_name);
260 if (it != var_map.end()) {return it->second;}
261
262 std::string normalized = "?" + std::to_string(i++);
263 var_map[old_name] = normalized;
264 return normalized;
265 }
266
267 void computeNormalizedDerived() const
268 {
269 if (normalizedDerivedComputed()) {return;}
270
271 normalized_predicate_ = this->predicate;
272 normalized_preconditions_ = this->preconditions;
273
274 std::map<std::string, std::string> var_map;
275 uint i = 0;
276
277 // Normalize predicate parameters
278 for (auto & p : normalized_predicate_.parameters) {
279 p.name = this->normalize_param(p.name, var_map, i);
280 }
281
282 // Normalize precondition nodes and their parameters
283 for (auto & node : normalized_preconditions_.nodes) {
284 if (node.node_type == plansys2_msgs::msg::Node::PARAMETER) {
285 node.name = this->normalize_param(node.name, var_map, i);
286 }
287
288 for (auto & p : node.parameters) {p.name = this->normalize_param(p.name, var_map, i);}
289 }
290
291 normalized_hash_ = 0;
292 std::hash_combine(normalized_hash_, std::hash_node(normalized_predicate_));
293 std::hash_combine(normalized_hash_, normalized_preconditions_.nodes.size());
294 for (auto & node : normalized_preconditions_.nodes) {
295 std::hash_combine(normalized_hash_, std::hash_node(node));
296 }
297 }
298
299 std::string toNormalizedString() const
300 {
301 if (!normalizedDerivedComputed()) {
302 computeNormalizedDerived();
303 }
304
305 std::ostringstream oss;
306
307 // Predicate
308 oss << "(" << normalized_predicate_.name;
309 for (const auto & p : normalized_predicate_.parameters) {
310 oss << " " << p.name;
311 }
312 oss << ")";
313
314 // Preconditions
315 if (!normalized_preconditions_.nodes.empty()) {
316 oss << " :- ";
317
318 bool first = true;
319 for (const auto & node : normalized_preconditions_.nodes) {
320 if (!first) {
321 oss << " ";
322 }
323 first = false;
324
325 oss << nodeToString(node);
326 }
327 }
328
329 return oss.str();
330 }
331
332private:
333 mutable plansys2_msgs::msg::Node normalized_predicate_;
334 mutable plansys2_msgs::msg::Tree normalized_preconditions_;
335 mutable bool normalized_cache_;
336 mutable std::size_t normalized_hash_;
337
338 static std::string
339 nodeToString(const plansys2_msgs::msg::Node & node)
340 {
341 std::ostringstream oss;
342
343 switch (node.node_type) {
344 case plansys2_msgs::msg::Node::AND:
345 oss << "(and";
346 for (const auto & p : node.parameters) {
347 oss << " " << p.name;
348 }
349 oss << ")";
350 break;
351
352 case plansys2_msgs::msg::Node::OR:
353 oss << "(or";
354 for (const auto & p : node.parameters) {
355 oss << " " << p.name;
356 }
357 oss << ")";
358 break;
359
360 case plansys2_msgs::msg::Node::PREDICATE:
361 oss << "(" << node.name;
362 for (const auto & p : node.parameters) {
363 oss << " " << p.name;
364 }
365 oss << ")";
366 break;
367
368 case plansys2_msgs::msg::Node::PARAMETER:
369 oss << node.name;
370 break;
371
372 default:
373 oss << "(unknown)";
374 break;
375 }
376
377 return oss.str();
378 }
