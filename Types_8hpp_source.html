35template<typename T>
36inline void hash_combine(std::size_t & seed, const T & value)
37{
38 seed ^= std::hash<T>{}(value) + 0x9e3779b9 + (seed << 6) + (seed >> 2);
39}
40
41inline std::size_t hash_node(const plansys2_msgs::msg::Node & node)
42{
43 std::size_t seed = 0;
44
45 plansys2::hash_combine(seed, node.name);
46 plansys2::hash_combine(seed, node.node_type);
47 plansys2::hash_combine(seed, node.children.size());
48 plansys2::hash_combine(seed, node.parameters.size());
49
50 for (const auto & param : node.parameters) {
51 plansys2::hash_combine(seed, param.name);
52 }
53 return seed;
54}

56template<typename UnorderedContainer>
57inline std::size_t unordered_container_hash(const UnorderedContainer & container)
58{
59 std::size_t h = 0;
60 for (const auto & elem : container) {
61 // Combine using XOR for order independence
62 h ^= std::hash<std::decay_t<decltype(elem)>>{}(elem);
63 }
64 // Optionally, mix in the size for more uniqueness
65 h ^= container.size();
66 return h;
67}
68
69// For unordered_map and similar
70template<typename Key, typename T, typename Hash, typename KeyEqual, typename Alloc>
71std::size_t unordered_container_hash(const std::unordered_map<Key, T, Hash, KeyEqual, Alloc> & map)
72{
73 std::size_t h = 0;
74 for (const auto & [key, value] : map) {
75 std::size_t entry_hash = std::hash<Key>{}(key);
76 entry_hash ^= std::hash<T>{}(value) + 0x9e3779b9;
77 h ^= entry_hash;
78 }
79 h ^= map.size();
80 return h;
81}

206class Derived : public plansys2_msgs::msg::Derived
207{
208public:
209 Derived()
210 : plansys2_msgs::msg::Derived() {computeNormalizedDerived();}
211 Derived(const plansys2_msgs::msg::Derived & derived) // NOLINT(runtime/explicit)
212 : plansys2_msgs::msg::Derived(derived)
213 {
214 computeNormalizedDerived();
215 }
216
217 bool operator==(const Derived & d) const
218 {
219 if (this == &d) {return true;}
220 if (!normalizedDerivedComputed()) {
221 computeNormalizedDerived();
222 }
223 if (!d.normalizedDerivedComputed()) {
224 d.computeNormalizedDerived();
225 }
226 return parser::pddl::checkNodeEquality(getNormalizedPredicate(), d.getNormalizedPredicate()) &&
227 parser::pddl::checkTreeEquality(
228 getNormalizedPreconditions(), d.getNormalizedPreconditions());
229 }
230
231 const size_t & getNormalizedHash() const {return normalized_hash_;}
232
233 const plansys2_msgs::msg::Node & getNormalizedPredicate() const {return normalized_predicate_;}
234
235 const plansys2_msgs::msg::Tree & getNormalizedPreconditions() const
236 {
237 return normalized_preconditions_;
238 }
239
240 Derived getNormalizedDerived()
241 {
242 if (!normalizedDerivedComputed()) {
243 computeNormalizedDerived();
244 }
245 Derived new_derived;
246 new_derived.predicate = normalized_predicate_;
247 new_derived.preconditions = normalized_preconditions_;
248 return new_derived;
249 }
250
251 bool normalizedDerivedComputed() const
252 {
253 return normalized_predicate_.name != "" && !normalized_preconditions_.nodes.empty();
254 }
255
256 std::string normalize_param(
257 const std::string & old_name, std::map<std::string, std::string> & var_map, uint & i) const
258 {
259 if (old_name.empty() || old_name.front() != '?') {return old_name;}
260
261 auto it = var_map.find(old_name);
262 if (it != var_map.end()) {return it->second;}
263
264 std::string normalized = "?" + std::to_string(i++);
265 var_map[old_name] = normalized;
266 return normalized;
267 }
268
269 void computeNormalizedDerived() const
270 {
271 if (normalizedDerivedComputed()) {return;}
272
273 normalized_predicate_ = this->predicate;
274 normalized_preconditions_ = this->preconditions;
275
276 std::map<std::string, std::string> var_map;
277 uint i = 0;
278
279 // Normalize predicate parameters
280 for (auto & p : normalized_predicate_.parameters) {
281 p.name = this->normalize_param(p.name, var_map, i);
282 }
283
284 // Normalize precondition nodes and their parameters
285 for (auto & node : normalized_preconditions_.nodes) {
286 if (node.node_type == plansys2_msgs::msg::Node::PARAMETER) {
287 node.name = this->normalize_param(node.name, var_map, i);
288 }
289
290 for (auto & p : node.parameters) {p.name = this->normalize_param(p.name, var_map, i);}
291 }
292
293 normalized_hash_ = 0;
294 plansys2::hash_combine(normalized_hash_, plansys2::hash_node(normalized_predicate_));
295 plansys2::hash_combine(normalized_hash_, normalized_preconditions_.nodes.size());
296 for (auto & node : normalized_preconditions_.nodes) {
297 plansys2::hash_combine(normalized_hash_, plansys2::hash_node(node));
298 }
299 }
300
301 std::string toNormalizedString() const
302 {
303 if (!normalizedDerivedComputed()) {
304 computeNormalizedDerived();
305 }
306
307 std::ostringstream oss;
308
309 // Predicate
310 oss << "(" << normalized_predicate_.name;
311 for (const auto & p : normalized_predicate_.parameters) {
312 oss << " " << p.name;
313 }
314 oss << ")";
315
316 // Preconditions
317 if (!normalized_preconditions_.nodes.empty()) {
318 oss << " :- ";
319
320 bool first = true;
321 for (const auto & node : normalized_preconditions_.nodes) {
322 if (!first) {
323 oss << " ";
324 }
325 first = false;
326
327 oss << nodeToString(node);
328 }
329 }
330
331 return oss.str();
332 }
333
334private:
335 mutable plansys2_msgs::msg::Node normalized_predicate_;
336 mutable plansys2_msgs::msg::Tree normalized_preconditions_;
337 mutable std::size_t normalized_hash_ {0};
338
339 static std::string
340 nodeToString(const plansys2_msgs::msg::Node & node)
341 {
342 std::ostringstream oss;
343
344 switch (node.node_type) {
345 case plansys2_msgs::msg::Node::AND:
346 oss << "(and";
347 for (const auto & p : node.parameters) {
348 oss << " " << p.name;
349 }
350 oss << ")";
351 break;
352
353 case plansys2_msgs::msg::Node::OR:
354 oss << "(or";
355 for (const auto & p : node.parameters) {
356 oss << " " << p.name;
357 }
358 oss << ")";
359 break;
360
361 case plansys2_msgs::msg::Node::PREDICATE:
362 oss << "(" << node.name;
363 for (const auto & p : node.parameters) {
364 oss << " " << p.name;
365 }
366 oss << ")";
367 break;
368
369 case plansys2_msgs::msg::Node::PARAMETER:
370 oss << node.name;
371 break;
372
373 default:
374 oss << "(unknown)";
375 break;
376 }
377
378 return oss.str();
379 }
