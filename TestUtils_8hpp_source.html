41class ExecutorThread
42{
43public:
44 explicit ExecutorThread(rclcpp::Executor & executor)
45 : executor_(executor),
46 thread_([this]() {
47 // spin(), not spin_some(): a MultiThreadedExecutor uses all its threads only there
48 while (!stop_ && rclcpp::ok()) {
49 executor_.spin();
50 }
51 finished_ = true;
52 })
53 {
54 }
55
56 ~ExecutorThread()
57 {
58 stop_ = true;
59 // Again until it ends: a cancel before spin() starts is lost
60 while (!finished_) {
61 executor_.cancel();
62 std::this_thread::sleep_for(std::chrono::milliseconds(1));
63 }
64 if (thread_.joinable()) {
65 thread_.join();
66 }
67 }
68
69 ExecutorThread(const ExecutorThread &) = delete;
70 ExecutorThread & operator=(const ExecutorThread &) = delete;
71
72private:
73 rclcpp::Executor & executor_;
74 std::atomic<bool> stop_ {false};
75 std::atomic<bool> finished_ {false};
76 std::thread thread_;
77};

127template<typename ServiceT>
128typename ServiceT::Response::SharedPtr call(
129 const rclcpp::Node::SharedPtr & spun_node, const std::string & name,
130 typename ServiceT::Request::SharedPtr request,
131 std::chrono::seconds timeout = std::chrono::seconds(5))
132{
133 auto client = spun_node->create_client<ServiceT>(name);
134 if (!client->wait_for_service(timeout)) {
135 return nullptr;
136 }
137 auto future = client->async_send_request(request);
138 if (future.wait_for(timeout) != std::future_status::ready) {
139 client->remove_pending_request(future);
140 return nullptr;
141 }
142 return future.get();
143}

152template<typename Condition>
153bool wait_for(
154 Condition condition, std::chrono::nanoseconds timeout = std::chrono::seconds(10))
155{
156 const auto end = std::chrono::steady_clock::now() + timeout;
157 while (!condition()) {
158 if (std::chrono::steady_clock::now() >= end || !rclcpp::ok()) {
159 return condition();
160 }
161 std::this_thread::sleep_for(std::chrono::milliseconds(10));
162 }
163 return true;
164}
