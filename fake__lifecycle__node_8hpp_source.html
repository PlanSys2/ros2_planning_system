35// Plain node that offers the lifecycle services of `name`, answering as told
36class FakeLifecycleNode : public rclcpp::Node
37{
38public:
39 explicit FakeLifecycleNode(const std::string & name)
40 : Node(name + "_fake")
41 {
42 get_state_ = create_service<lifecycle_msgs::srv::GetState>(
43 name + "/get_state",
44 [this](
45 const std::shared_ptr<lifecycle_msgs::srv::GetState::Request>,
46 std::shared_ptr<lifecycle_msgs::srv::GetState::Response> response) {
47 get_state_calls++;
48 std::this_thread::sleep_for(get_state_delay.load());
49 response->current_state.id = state;
50 });
51 change_state_ = create_service<lifecycle_msgs::srv::ChangeState>(
52 name + "/change_state",
53 [this](
54 std::shared_ptr<rclcpp::Service<lifecycle_msgs::srv::ChangeState>> service,
55 std::shared_ptr<rmw_request_id_t> header,
56 const std::shared_ptr<lifecycle_msgs::srv::ChangeState::Request>) {
57 std::this_thread::sleep_for(change_state_delay.load());
58 if (on_change_state) {
59 on_change_state();
60 }
61 // As DDS may drop it while the nodes are still discovering each other
62 if (lose_change_state_response) {
63 return;
64 }
65 lifecycle_msgs::srv::ChangeState::Response response;
66 response.success = change_state_success;
67 service->send_response(*header, response);
68 });
69 }
70
71 std::atomic<uint8_t> state {lifecycle_msgs::msg::State::PRIMARY_STATE_UNCONFIGURED};
72 std::atomic<bool> change_state_success {true};
73 std::atomic<bool> lose_change_state_response {false};
74 std::atomic<std::chrono::milliseconds> get_state_delay {std::chrono::milliseconds(0)};
75 std::atomic<std::chrono::milliseconds> change_state_delay {std::chrono::milliseconds(0)};
76 std::function<void()> on_change_state;
77 std::atomic<int> get_state_calls {0};
78
79private:
80 rclcpp::Service<lifecycle_msgs::srv::GetState>::SharedPtr get_state_;
81 rclcpp::Service<lifecycle_msgs::srv::ChangeState>::SharedPtr change_state_;
82};

84// Lifecycle node whose configure or activate transition can fail
85class FailingLifecycleNode : public rclcpp_lifecycle::LifecycleNode
86{
87public:
88 using CallbackReturn = rclcpp_lifecycle::node_interfaces::LifecycleNodeInterface::CallbackReturn;
89
90 FailingLifecycleNode(const std::string & name, bool fail_configure, bool fail_activate)
91 : LifecycleNode(name), fail_configure_(fail_configure), fail_activate_(fail_activate) {}
92
93 CallbackReturn on_configure(const rclcpp_lifecycle::State &) override
94 {
95 return fail_configure_ ? CallbackReturn::FAILURE : CallbackReturn::SUCCESS;
96 }
97
98 CallbackReturn on_activate(const rclcpp_lifecycle::State &) override
99 {
100 return fail_activate_ ? CallbackReturn::FAILURE : CallbackReturn::SUCCESS;
101 }
102
103private:
104 bool fail_configure_;
105 bool fail_activate_;
106};
