33// Plain node that offers the lifecycle services of `name`, answering as told
34class FakeLifecycleNode : public rclcpp::Node
35{
36public:
37 explicit FakeLifecycleNode(const std::string & name)
38 : Node(name + "_fake")
39 {
40 get_state_ = create_service<lifecycle_msgs::srv::GetState>(
41 name + "/get_state",
42 [this](
43 const std::shared_ptr<lifecycle_msgs::srv::GetState::Request>,
44 std::shared_ptr<lifecycle_msgs::srv::GetState::Response> response) {
45 get_state_calls++;
46 std::this_thread::sleep_for(get_state_delay);
47 response->current_state.id = state;
48 });
49 change_state_ = create_service<lifecycle_msgs::srv::ChangeState>(
50 name + "/change_state",
51 [this](
52 std::shared_ptr<rclcpp::Service<lifecycle_msgs::srv::ChangeState>> service,
53 std::shared_ptr<rmw_request_id_t> header,
54 const std::shared_ptr<lifecycle_msgs::srv::ChangeState::Request>) {
55 std::this_thread::sleep_for(change_state_delay);
56 if (on_change_state) {
57 on_change_state();
58 }
59 // As DDS may drop it while the nodes are still discovering each other
60 if (lose_change_state_response) {
61 return;
62 }
63 lifecycle_msgs::srv::ChangeState::Response response;
64 response.success = change_state_success;
65 service->send_response(*header, response);
66 });
67 }
68
69 std::atomic<uint8_t> state {lifecycle_msgs::msg::State::PRIMARY_STATE_UNCONFIGURED};
70 std::atomic<bool> change_state_success {true};
71 std::atomic<bool> lose_change_state_response {false};
72 std::chrono::milliseconds get_state_delay {0};
73 std::chrono::milliseconds change_state_delay {0};
74 std::function<void()> on_change_state;
75 std::atomic<int> get_state_calls {0};
76
77private:
78 rclcpp::Service<lifecycle_msgs::srv::GetState>::SharedPtr get_state_;
79 rclcpp::Service<lifecycle_msgs::srv::ChangeState>::SharedPtr change_state_;
80};

106// Spins an executor in its own thread while it lives
107class Spinner
108{
109public:
110 Spinner()
111 : exe_(std::make_shared<rclcpp::executors::MultiThreadedExecutor>(
112 rclcpp::ExecutorOptions(), 4)) {}
113
114 ~Spinner()
115 {
116 exe_->cancel();
117 if (thread_.joinable()) {
118 thread_.join();
119 }
120 }
121
122 // Keeps the node alive until the executor stops
123 template<typename NodeT>
124 void add(std::shared_ptr<NodeT> node)
125 {
126 keep_alive_.push_back(node);
127 exe_->add_node(node->get_node_base_interface());
128 }
129
130 void start()
131 {
132 thread_ = std::thread([this]() {exe_->spin();});
133 }
134
135private:
136 std::shared_ptr<rclcpp::executors::MultiThreadedExecutor> exe_;
137 std::thread thread_;
138 std::vector<std::shared_ptr<void>> keep_alive_;
139};
