335template<typename NodeEdgesMap>
336std::size_t unordered_nodeedgesmap_hash(const NodeEdgesMap & map)
337{
338 std::size_t seed = 0;
339 for (const auto & [key, neighbors] : map) {
340 // Hash the key
341 std::size_t entry_hash = std::hash<std::decay_t<decltype(key)>>{}(key);
342 // Hash the neighbor set (order-independent)
343 entry_hash ^= unordered_container_hash(neighbors);
344 // XOR with seed (order-independent for map)
345 seed ^= entry_hash;
346 }
347 // Mix in size for extra entropy
348 seed ^= map.size();
349 return seed;
350}

352template<>
353struct hash<plansys2::DerivedResolutionGraph>
354{
355 std::size_t operator()(const plansys2::DerivedResolutionGraph & graph) const noexcept
356 {
357 std::size_t seed = 0;
358 hash_combine(seed, graph.edge_count_);
359 seed ^= unordered_container_hash(graph.nodes_);
360 seed ^= unordered_container_hash(graph.roots_);
361 seed ^= unordered_container_hash(graph.derived_predicates_);
362 seed ^= unordered_container_hash(graph.actions_);
363 seed ^= unordered_container_hash(graph.node_ids_);
364 seed ^= unordered_nodeedgesmap_hash(graph.adj_list_);
365 seed ^= unordered_nodeedgesmap_hash(graph.parent_nodes_);
366 return seed;
367 }
368};
