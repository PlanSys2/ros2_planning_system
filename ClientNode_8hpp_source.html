151 template<typename FutureT>
152 rclcpp::FutureReturnCode wait(FutureT & future, std::chrono::nanoseconds timeout)
153 {
154 const auto end = std::chrono::steady_clock::now() + timeout;
155 while (running()) {
156 const auto left = end - std::chrono::steady_clock::now();
157 if (left <= std::chrono::nanoseconds::zero()) {
158 return future.wait_for(std::chrono::seconds(0)) == std::future_status::ready ?
159 rclcpp::FutureReturnCode::SUCCESS : rclcpp::FutureReturnCode::TIMEOUT;
160 }
161 // In slices, to see a shutdown
162 const auto slice = std::min<std::chrono::nanoseconds>(left, kSlice);
163 std::unique_lock<std::mutex> lock(serve_mutex_, std::try_to_lock);
164 if (lock.owns_lock()) {
165 if (executor_->spin_until_future_complete(future, slice) ==
166 rclcpp::FutureReturnCode::SUCCESS)
167 {
168 return rclcpp::FutureReturnCode::SUCCESS;
169 }
170 } else if (future.wait_for(slice) == std::future_status::ready) {
171 // Another thread of the client serves the group, and brought it
172 return rclcpp::FutureReturnCode::SUCCESS;
173 }
174 }
175 return rclcpp::FutureReturnCode::INTERRUPTED;
176 }
