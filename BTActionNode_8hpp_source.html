96 bool createActionClient(const std::string & action_name)
97 {

202 BT::NodeStatus tick() override
203 {
204 switch (state_) {
205 case IDLE:
206 {
207 RCLCPP_DEBUG(node_->get_logger(), "%s IDLE", node_->get_name());
208 assert((status() == BT::NodeStatus::IDLE));
209
210 if (!createActionClient(action_name_)) {
211 RCLCPP_ERROR(node_->get_logger(), "Failed to create action client");
212 return BT::NodeStatus::FAILURE;
213 }
214
215 // User defined tick
216 auto user_status = on_tick();
217 if (user_status != BT::NodeStatus::RUNNING) {
218 return user_status;
219 }
220
221 on_new_goal_received();
222
223 state_ = GOAL_SENT;
224
225 return BT::NodeStatus::RUNNING;
226 }
227 break;
228
229 case GOAL_SENT:
230 {
231 RCLCPP_DEBUG(node_->get_logger(), "%s GOAL_SENT", node_->get_name());
232 if (future_goal_handle_) {
233 auto elapsed =
234 (node_->now() - goal_sent_ts_).template to_chrono<std::chrono::milliseconds>();
235 if (!is_future_goal_handle_complete(elapsed)) {
236 RCLCPP_ERROR(
237 node_->get_logger(),
238 "Goal was rejected by action server %s", action_name_.c_str());
239 state_ = GOAL_FAILURE;
240 future_goal_handle_.reset();
241 return BT::NodeStatus::FAILURE;
242 } else {
243 state_ = GOAL_EXECUTING;
244 return BT::NodeStatus::RUNNING;
245 }
246 } else {
247 if ((node_->now() - goal_sent_ts_) > server_timeout_) {
248 RCLCPP_ERROR(
249 node_->get_logger(),
250 "Failed to send goal to action server %s", action_name_.c_str());
251 state_ = GOAL_FAILURE;
252 future_goal_handle_.reset();
253 return BT::NodeStatus::FAILURE;
254 } else {
255 return BT::NodeStatus::RUNNING;
256 }
257 }
258 }
259 break;
260
261 case GOAL_EXECUTING:
262 {
263 RCLCPP_DEBUG(node_->get_logger(), "%s GOAL_EXECUTING", node_->get_name());
264 auto goal_status = goal_handle_->get_status();
265 auto user_status = on_tick();
266
267 if (user_status != BT::NodeStatus::RUNNING) {
268 cancel_goal();
269 state_ = GOAL_CANCELLING;
270 return user_status;
271 }
272
273 if (goal_updated_ && (goal_status == action_msgs::msg::GoalStatus::STATUS_EXECUTING ||
274 goal_status == action_msgs::msg::GoalStatus::STATUS_ACCEPTED))
275 {
276 on_new_goal_received();
277 state_ = GOAL_SENT;
278 }
279
280 callback_group_executor_.spin_some();
281
282 if (goal_result_available_) {
283 state_ = GOAL_FINISHING;
284 }
285
286 return BT::NodeStatus::RUNNING;
287 }
288 break;
289
290 case GOAL_FINISHING:
291 {
292 RCLCPP_DEBUG(node_->get_logger(), "%s GOAL_FINISHING", node_->get_name());
293 switch (result_.code) {
294 case rclcpp_action::ResultCode::SUCCEEDED:
295 state_ = GOAL_FINISHED;
296 return on_success();
297
298 case rclcpp_action::ResultCode::ABORTED:
299 state_ = GOAL_FINISHED;
300 return on_aborted();
301
302 case rclcpp_action::ResultCode::CANCELED:
303 state_ = GOAL_FINISHED;
304 return on_cancelled();
305
306 default:
307 throw std::logic_error("BtActionNode::Tick: invalid status value");
308 }
309 }
310 break;
311
312 case GOAL_CANCELLING:
313 {
314 RCLCPP_DEBUG(node_->get_logger(), "%s GOAL_CANCELLING", node_->get_name());
315 if (future_cancel_handle_.valid()) {
316 state_ = GOAL_FINISHED;
317 return BT::NodeStatus::SUCCESS;
318 } else {
319 RCLCPP_ERROR(
320 node_->get_logger(),
321 "Failed to cancel action server for %s", action_name_.c_str());
322 state_ = GOAL_FAILURE;
323 future_goal_handle_.reset();
324 return BT::NodeStatus::FAILURE;
325 }
326 }
327 break;
328
329 case GOAL_FINISHED:
330 {
331 RCLCPP_DEBUG(node_->get_logger(), "%s GOAL_FINISHED", node_->get_name());
332 state_ = IDLE;
333 return BT::NodeStatus::SUCCESS;
334 }
335 break;
336
337 case GOAL_FAILURE:
338 {
339 RCLCPP_DEBUG(node_->get_logger(), "%s GOAL_FAILURE", node_->get_name());
340 state_ = IDLE;
341 return BT::NodeStatus::FAILURE;
342 }
343 break;
344
345 default:
346 break;
347 }
348
349 goal_handle_.reset();
350 return BT::NodeStatus::RUNNING;
351 }

370 void cancel_goal()
371 {
372 if (goal_handle_) {
373 future_cancel_handle_ = action_client_->async_cancel_goal(goal_handle_);
374 if (callback_group_executor_.spin_until_future_complete(future_cancel_handle_,
375 server_timeout_) != rclcpp::FutureReturnCode::SUCCESS)
376 {
377 RCLCPP_ERROR(
378 node_->get_logger(), "Failed to cancel action server for %s", action_name_.c_str());
379 }
380
381 auto future_result = action_client_->async_get_result(goal_handle_);
382 if (callback_group_executor_.spin_until_future_complete(future_result, server_timeout_) !=
383 rclcpp::FutureReturnCode::SUCCESS)
384 {
385 RCLCPP_ERROR(
386 node_->get_logger(),
387 "Failed to get result for %s in node halt!", action_name_.c_str());
388 }
389
390 on_cancelled();
391 } else {
392 RCLCPP_WARN(
393 node_->get_logger(),
394 "Cannot cancel goal for %s: goal handle is null", action_name_.c_str());
395 }

402 bool should_cancel_goal()
403 {
404 // Shut the node down if it is currently running
405 if (status() != BT::NodeStatus::RUNNING) {
406 return false;
407 }
408
409 // No need to cancel the goal if goal handle is invalid
410 if (!goal_handle_) {
411 return false;
412 }
413
414 callback_group_executor_.spin_some();
415 auto status = goal_handle_->get_status();
416
417 // Check if the goal is still executing
418 return status == action_msgs::msg::GoalStatus::STATUS_ACCEPTED ||
419 status == action_msgs::msg::GoalStatus::STATUS_EXECUTING;
420 }

425 void on_new_goal_received()
426 {
427 goal_result_available_ = false;
428 auto send_goal_options = typename rclcpp_action::Client<ActionT>::SendGoalOptions();
429 send_goal_options.result_callback =
430 [this](const typename rclcpp_action::ClientGoalHandle<ActionT>::WrappedResult & result) {
431 // TODO(#1652): a work around until rcl_action interface is updated
432 // if goal ids are not matched, the older goal call this callback so ignore the result
433 // if matched, it must be processed (including aborted)
434 if (this->goal_handle_->get_goal_id() == result.goal_id) {
435 goal_result_available_ = true;
436 result_ = result;
437 }
438 };
439 send_goal_options.feedback_callback =
440 [this](typename rclcpp_action::ClientGoalHandle<ActionT>::SharedPtr,
441 const std::shared_ptr<const typename ActionT::Feedback> feedback) {
442 on_feedback(feedback);
443 };
444
445 RCLCPP_INFO(
446 node_->get_logger(), "Sending goal to action server %s", action_name_.c_str());
447
448 future_goal_handle_ = std::make_shared<
449 std::shared_future<typename rclcpp_action::ClientGoalHandle<ActionT>::SharedPtr>>(
450 action_client_->async_send_goal(goal_, send_goal_options));
451 goal_sent_ts_ = node_->now();
452 }

460 bool is_future_goal_handle_complete(std::chrono::milliseconds & elapsed)
461 {
462 auto remaining = server_timeout_ - elapsed;
463
464 // server has already timed out, no need to sleep
465 if (remaining <= std::chrono::milliseconds(0)) {
466 future_goal_handle_.reset();
467 return false;
468 }
469
470 auto timeout = remaining > max_timeout_ ? max_timeout_ : remaining;
471 auto result =
472 callback_group_executor_.spin_until_future_complete(*future_goal_handle_, timeout);
473 elapsed += timeout;
474
475 if (result == rclcpp::FutureReturnCode::INTERRUPTED) {
476 future_goal_handle_.reset();
477 throw std::runtime_error("send_goal failed");
478 }
479
480 if (result == rclcpp::FutureReturnCode::SUCCESS) {
481 goal_handle_ = future_goal_handle_->get();
482 future_goal_handle_.reset();
483 if (!goal_handle_) {
484 throw std::runtime_error("Goal was rejected by the action server");
485 }
486 return true;
487 }
488
489 return false;
490 }
