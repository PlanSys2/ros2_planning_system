117// Throws if a name appears twice in a list read from PDDL
118inline void assertUnique(const TokenStruct<std::string> & ts, const std::string & what)
119{
120 if (ts.tokenMap.size() == ts.size()) {
121 return;
122 }
123 std::set<std::string> seen;
124 for (unsigned i = 0; i < ts.size(); ++i) {
125 if (!seen.insert(ts[i]).second) {
126 throw std::runtime_error("Duplicate " + what + " \"" + ts[i] + "\"");
127 }
128 }
129}

131// The names in the scope of a quantifier body: those outside, then its variables, which
132// hide outer ones with the same name
133inline TokenStruct<std::string> quantifierScope(
134 const TokenStruct<std::string> & outer, const TokenStruct<std::string> & vars)
135{
136 TokenStruct<std::string> scope(outer);
137 for (unsigned i = 0; i < vars.size(); ++i) {
138 scope.tokenMap[vars[i]] = scope.size();
139 scope.tokens.push_back(vars[i]);
140 }
141 scope.types.insert(scope.types.end(), vars.types.begin(), vars.types.end());
142 return scope;
143}
144
145// The same scope when printing: base names from outside, then the variables
146inline TokenStruct<std::string> printScope(
147 const TokenStruct<std::string> & outer, unsigned base, const std::vector<std::string> & names)
148{
149 TokenStruct<std::string> scope;
150 for (unsigned i = 0; i < base; ++i) {
151 scope.tokens.push_back(i < outer.size() ? outer[i] : "?" + std::to_string(i));
152 }
153 for (const auto & name : names) {
154 scope.tokens.push_back(name);
155 }
156 for (unsigned i = 0; i < scope.size(); ++i) {
157 scope.tokenMap[scope.tokens[i]] = i;
158 }
159 return scope;
160}
