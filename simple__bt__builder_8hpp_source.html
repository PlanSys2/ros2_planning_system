69struct ActionGraph
70{
71 using Ptr = std::shared_ptr<ActionGraph>;
72 static Ptr make_shared() {return std::make_shared<ActionGraph>();}
73
74 std::list<ActionNode::Ptr> roots;
75 std::map<double, std::list<ActionNode::Ptr>> levels;
76
77 // Arcs point both ways, so the nodes would keep each other alive (#430)
78 ~ActionGraph()
79 {
80 auto unlink = [](const std::list<ActionNode::Ptr> & nodes) {
81 for (const auto & node : nodes) {
82 node->in_arcs.clear();
83 node->out_arcs.clear();
84 }
85 };
86 unlink(roots);
87 for (const auto & level : levels) {
88 unlink(level.second);
89 }
90 }
91};
