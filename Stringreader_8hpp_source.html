75 std::vector<std::string> getLines(const std::string & text)
76 {
77 std::vector<std::string> ret;
78 size_t start = 0, end = 0;
79
80 while (end != std::string::npos) {
81 end = text.find("\n", start);
82 ret.push_back(
83 text.substr(start, (end == std::string::npos) ? std::string::npos : end - start));

126 // get token converted to lowercase
127 std::string getToken()
128 {
129 std::ostringstream os;
130 while (c < s.size() && !ignore(s[c]) && !paren(s[c]) && s[c] != ',') {
131 os << (65 <= s[c] && s[c] <= 90 ? static_cast<char>(s[c++] + 32) : s[c++]);
132 }
133
134 return os.str();
135 }

147 // assert syntax
148 void assert_token(const std::string & t)
149 {
150 unsigned b = 0;
151 for (unsigned k = 0; c + k < s.size() && k < t.size(); ++k) {
152 b += s[c + k] == t[k] || (65 <= s[c + k] && s[c + k] <= 90 && s[c + k] == t[k] - 32);
153 }
154
155 if (b < t.size()) {
156 printLine();
157 throw ExpectedToken(t);
158 }
159 c += t.size();
160 next();
161 }

178 // parse a typed list
179 // if check is true, checks that types exist
180 TokenStruct<std::string> parseTypedList(
181 bool check, const TokenStruct<Type *> & ts = TokenStruct<Type *>(), const std::string & lt = "")
182 {
183 unsigned k = 0;
184 TokenStruct<std::string> out;
185 for (next(); getChar() != ')' && lt.find(getChar()) == std::string::npos; next()) {
186 if (getChar() == '-') {
187 assert_token("-");
188
189 std::string t;
190 // check if the type is "either"
191 if (getChar() == '(') {
192 assert_token("(");
193 assert_token("either");
194
195 t = "( either";
196 for (; getChar() != ')'; next()) {
197 if (check) {
198 t += " " + getToken(ts);
199 } else {
200 t += " " + getToken();
201 }
202 }
203 t += " )";
204 ++c;
205 } else if (check) {
206 t = getToken(ts);
207 } else {
208 t = getToken();
209 }
210
211 out.types.insert(out.types.end(), out.size() - k, t);
212 k = out.size();
213 } else if (getChar() == '(') {
214 assert_token("(");
215 assert_token(":private");
216 getToken();
217 out.append(parseTypedList(check, ts));
218 } else {
219 out.insert(getToken());
220 }
221 }
222 if (k < out.size()) {out.types.insert(out.types.end(), out.size() - k, check ? "object" : "");}
223 ++c;
224
225 return out;
226 }
