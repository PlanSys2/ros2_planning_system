65 bool operator==(const NodeVariant & other) const
66 {
67 if (this == &other) {
68 return true; // Same instance
69 }
70 // Check type first
71 if (this->getNodeType() != other.getNodeType()) {return false;}
72
73 // Predicate case
74 if (this->isPredicate()) {
75 return parser::pddl::checkNodeEquality(
76 this->getPredicateNode(), other.getPredicateNode(), false);
77 }
78
79 // Function case
80 if (this->isFunction()) {
81 return parser::pddl::checkNodeEquality(
82 this->getFunctionNode(), other.getFunctionNode(), false);
83 }
84
85 return *node_ == *other.node_;
86 }

92 std::string getNodeName() const
93 {
94 std::string node_name;
95 if (std::holds_alternative<plansys2::Predicate>(*node_)) {
96 node_name = std::get<plansys2::Predicate>(*node_).name;
97 } else if (std::holds_alternative<plansys2::Function>(*node_)) {
98 node_name = std::get<plansys2::Function>(*node_).name;
99 } else if (std::holds_alternative<plansys2::Derived>(*node_)) {
100 node_name = std::get<plansys2::Derived>(*node_).predicate.name;
101 } else if (std::holds_alternative<plansys2::ActionVariant>(*node_)) {
102 node_name = std::get<plansys2::ActionVariant>(*node_).get_action_name();
103 }

143 std::string getNodeType() const
144 {
145 if (std::holds_alternative<plansys2::Predicate>(*node_)) {
146 return "predicate";
147 } else if (std::holds_alternative<plansys2::Function>(*node_)) {
148 return "function";
149 } else if (std::holds_alternative<plansys2::Derived>(*node_)) {
150 return "derived";
151 } else if (std::holds_alternative<plansys2::ActionVariant>(*node_)) {
152 return std::get<plansys2::ActionVariant>(*node_).is_action() ? "action" : "durative action";

194inline plansys2::NodeVariant nodeMsgToVariant(const plansys2_msgs::msg::Node & node_msg)
195{
196 if (node_msg.node_type == plansys2_msgs::msg::Node::PREDICATE) {
197 return plansys2::Predicate(node_msg);
198 } else if (node_msg.node_type == plansys2_msgs::msg::Node::FUNCTION) {
199 return plansys2::Function(node_msg);
200 } else {
201 throw std::runtime_error("Unsupported node_type for NodeVariant conversion");
202 }
203}
204
205inline bool operator==(const NodeVariant & lhs, const plansys2_msgs::msg::Node & rhs)
206{
207 if (lhs.isPredicate()) {
208 return lhs.getPredicateNode() == static_cast<plansys2::Predicate>(rhs);
209 } else if (lhs.isFunction()) {
210 return lhs.getFunctionNode() == static_cast<plansys2::Function>(rhs);
211 }
212 return false;
213}

215inline std::ostream & operator<<(std::ostream & os, const NodeVariant & node)
216{
217 os << node.getNodeType() << ":" << node.getNodeName();
218
219 // Print parameters, if available
220 if (node.isPredicate()) {
221 for (const auto & param : node.getPredicateNode().parameters) {
222 os << " " << param.name;
223 }
224 } else if (node.isFunction()) {
225 for (const auto & param : node.getFunctionNode().parameters) {
226 os << " " << param.name;
227 }
228 } else if (node.isDerived()) {
229 const auto & pred = node.getDerivedPredicate();
230 for (const auto & param : pred.parameters) {
231 os << " " << param.name;
232 }
233 } else if (node.isActionVariant()) {
234 const auto & action = node.getActionVariantNode();
235 for (const auto & param : action.get_action_params()) {
236 os << " " << param.name;
237 }
238 }
239 return os;
240}

247inline std::size_t hash_node_variant(const plansys2_msgs::msg::Node & node)
248{
249 std::size_t seed = 0;
250
251 hash_combine(seed, node.name);
252 hash_combine(seed, node.node_type);
253 hash_combine(seed, node.parameters.size());
254
255 return seed;
256}

264 return hash_node_variant(nv.getPredicateNode());
265 } else if (nv.isFunction()) {
266 return hash_node_variant(nv.getFunctionNode());
267 }
268 return nv.hash();
