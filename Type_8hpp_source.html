101 std::pair<bool, int> parseConstant(const std::string & object)
102 {
103 int k = 0;
104
105 int i = constants.index(object);
106 if (i < 0) {
107 k += constants.size();
108 } else {
109 return std::make_pair(true, -1 - i);
110 }
111
112 for (unsigned i = 0; i < subtypes.size(); ++i) {
113 std::pair<bool, int> p = subtypes[i]->parseConstant(object);
114 if (p.first) {
115 return std::make_pair(true, -k + p.second);
116 } else {
117 k += p.second;
118 }
119 }
120
121 return std::make_pair(false, k);
122 }

129 if (i < 0) {
130 k += objects.size();
131 } else {
132 return std::make_pair(true, i);
133 }
134
135 for (unsigned i = 0; i < subtypes.size(); ++i) {
136 std::pair<bool, unsigned> p = subtypes[i]->parseObject(object);
137 if (p.first) {

140 k += p.second;

147 std::pair<std::string, int> object(int index)
148 {
149 if (index < 0) {
150 if (-index <= static_cast<int>(constants.size())) {
151 return std::make_pair(constants[-1 - index], 0);
152 } else {
153 index += constants.size();
154 }
155 } else {
156 if (index < static_cast<int>(objects.size())) {
157 return std::make_pair(objects[index], 0);
158 } else {
159 index -= objects.size();
160 }
161 }
162
163 for (unsigned i = 0; i < subtypes.size(); ++i) {
164 std::pair<std::string, int> p = subtypes[i]->object(index);
165 if (p.first.size()) {
166 return p;
167 } else {
168 index = p.second;
169 }
170 }
171
172 return std::make_pair("", index);
173 }
