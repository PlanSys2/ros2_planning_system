50class CompositeExpression : public Expression
51{
52public:
53 std::string op;
54 Expression * left;
55 Expression * right;
56
57 explicit CompositeExpression(const std::string & c)
58 : op(c) {}
59
60 CompositeExpression(const std::string & c, Expression * l, Expression * r)
61 : op(c), left(l), right(r)
62 {
63 }
64
65 ~CompositeExpression()
66 {
67 delete left;
68 delete right;
69 }
70
71 void parse(Stringreader & f, TokenStruct<std::string> & ts, Domain & d)
72 {
73 f.next();
74 left = createExpression(f, ts, d);
75 right = createExpression(f, ts, d);
76 f.next();
77 f.assert_token(")");
78 }
79
80 std::string info() const
81 {
82 std::ostringstream os;
83 os << "(" << op << " " << left->info() << " " << right->info() << ")";
84 return os.str();
85 }
86
87 void PDDLPrint(
88 std::ostream & s, unsigned indent, const TokenStruct<std::string> & ts,
89 const Domain & d) const override
90 {
91 tabindent(s, indent);
92 s << "( " << op << " ";
93 left->PDDLPrint(s, indent, ts, d);
94 s << " ";
95 right->PDDLPrint(s, indent, ts, d);
96 s << " )";
97 }
98
99 plansys2_msgs::msg::Node::SharedPtr getTree(
100 plansys2_msgs::msg::Tree & tree, const Domain & d,
101 const std::vector<std::string> & replace = {}) const override
102 {
103 plansys2_msgs::msg::Node::SharedPtr node = std::make_shared<plansys2_msgs::msg::Node>();
104 node->node_type = plansys2_msgs::msg::Node::EXPRESSION;
105 node->expression_type = getExprType(op);
106 node->node_id = tree.nodes.size();
107 tree.nodes.push_back(*node);
108
109 plansys2_msgs::msg::Node::SharedPtr left_child = left->getTree(tree, d, replace);
110 tree.nodes[node->node_id].children.push_back(left_child->node_id);
111
112 plansys2_msgs::msg::Node::SharedPtr right_child = right->getTree(tree, d, replace);
113 tree.nodes[node->node_id].children.push_back(right_child->node_id);
114
115 return node;
116 }
117
118 double compute(double x, double y)
119 {
120 double res = 0;
121
122 if (op == "+") {
123 res = x + y;
124 } else if (op == "-") {
125 res = x - y;
126 } else if (op == "*") {
127 res = x * y;
128 } else if (op == "/") {
129 res = (y == 0 ? 0 : x / y);
130 }
131
132 return res;
133 }
134
135 double evaluate() {return compute(left->evaluate(), right->evaluate());}
136
137 double evaluate(Instance & ins, const StringVec & par)
138 {
139 return compute(left->evaluate(ins, par), right->evaluate(ins, par));
140 }
141
142 IntSet params()
143 {
144 IntSet lpars = left->params();
145 IntSet rpars = right->params();
146 lpars.insert(rpars.begin(), rpars.end());
147 return lpars;
148 }
149
150 Condition * copy(Domain & d)
151 {
152 Expression * cleft = dynamic_cast<Expression *>(left->copy(d));
153 Expression * cright = dynamic_cast<Expression *>(right->copy(d));
154 return new CompositeExpression(op, cleft, cright);
155 }
156};

298 {
299 plansys2_msgs::msg::Node::SharedPtr node = std::make_shared<plansys2_msgs::msg::Node>();
300 node->node_type = plansys2_msgs::msg::Node::PARAMETER;
301 node->node_id = tree.nodes.size();
302 node->name = "?" + std::to_string(param);
303 plansys2_msgs::msg::Param node_param;
304 if (static_cast<size_t>(replace.size()) > static_cast<size_t>(param)) {
305 node_param.name = replace[param];
306 } else {
307 node_param.name = "?" + std::to_string(param);
308 }
309 node->parameters.push_back(node_param);
310 tree.nodes.push_back(*node);
311 return node;
312 }
