54// (op a b ...): comparisons, - and / take two operands, + and * two or more, and - also one
55class CompositeExpression : public Expression
56{
57public:
58 std::string op;
59 std::vector<Expression *> args;
60
61 explicit CompositeExpression(const std::string & c)
62 : op(c) {}
63
64 CompositeExpression(const std::string & c, Expression * l, Expression * r)
65 : op(c), args({l, r})
66 {
67 }
68
69 CompositeExpression(const std::string & c, std::vector<Expression *> a)
70 : op(c), args(std::move(a))
71 {
72 }
73
74 ~CompositeExpression()
75 {
76 for (auto arg : args) {delete arg;}
77 }
78
79 void parse(Stringreader & f, TokenStruct<std::string> & ts, Domain & d)
80 {
81 for (f.next(); f.getChar() != ')'; f.next()) {
82 std::unique_ptr<Expression> arg(createExpression(f, ts, d));
83 if (!arg) {
84 throw std::runtime_error("Missing operand of " + op);
85 }
86 args.push_back(arg.release());
87 }
88 ++f.c;
89
90 const bool multi = op == "+" || op == "*";
91 const bool fits = multi ? args.size() >= 2 :
92 (op == "-" ? args.size() == 1 || args.size() == 2 : args.size() == 2);
93 if (!fits) {
94 throw std::runtime_error(
95 "Wrong number of operands (" + std::to_string(args.size()) + ") for " + op);
96 }
97 }
98
99 std::string info() const
100 {
101 std::ostringstream os;
102 os << "(" << op;
103 for (auto arg : args) {os << " " << arg->info();}
104 os << ")";
105 return os.str();
106 }
107
108 void PDDLPrint(
109 std::ostream & s, unsigned indent, const TokenStruct<std::string> & ts,
110 const Domain & d) const override
111 {
112 tabindent(s, indent);
113 s << "( " << op;
114 for (auto arg : args) {
115 s << " ";
116 arg->PDDLPrint(s, 0, ts, d);
117 }
118 s << " )";
119 }

121 // In the tree every operation is binary: (+ a b c) is (+ (+ a b) c) and (- a) is (- 0 a)
122 plansys2_msgs::msg::Node::SharedPtr getTree(
123 plansys2_msgs::msg::Tree & tree, const Domain & d,
124 const std::vector<std::string> & replace = {}) const override
125 {
126 return binaryTree(tree, d, replace, args.size());
127 }
128
129 double compute(double x, double y)
130 {
131 double res = 0;
132
133 if (op == "+") {
134 res = x + y;
135 } else if (op == "-") {
136 res = x - y;
137 } else if (op == "*") {
138 res = x * y;
139 } else if (op == "/") {
140 res = (y == 0 ? 0 : x / y);
141 }
142
143 return res;
144 }
145
146 double evaluate()
147 {
148 if (args.size() == 1) {return compute(0, args[0]->evaluate());}
149 double res = args[0]->evaluate();
150 for (size_t i = 1; i < args.size(); ++i) {res = compute(res, args[i]->evaluate());}
151 return res;
152 }
153
154 double evaluate(Instance & ins, const StringVec & par)
155 {
156 if (args.size() == 1) {return compute(0, args[0]->evaluate(ins, par));}
157 double res = args[0]->evaluate(ins, par);
158 for (size_t i = 1; i < args.size(); ++i) {res = compute(res, args[i]->evaluate(ins, par));}
159 return res;
160 }
161
162 IntSet params()
163 {
164 IntSet pars;
165 for (auto arg : args) {
166 IntSet arg_pars = arg->params();
167 pars.insert(arg_pars.begin(), arg_pars.end());
168 }
169 return pars;
170 }
171
172 Condition * copy(Domain & d)
173 {
174 std::vector<Expression *> copies;
175 for (auto arg : args) {copies.push_back(dynamic_cast<Expression *>(arg->copy(d)));}
176 return new CompositeExpression(op, copies);
177 }
178
179private:
180 // The operation on the first n operands
181 plansys2_msgs::msg::Node::SharedPtr binaryTree(
182 plansys2_msgs::msg::Tree & tree, const Domain & d, const std::vector<std::string> & replace,
183 size_t n) const
184 {
185 plansys2_msgs::msg::Node::SharedPtr node = std::make_shared<plansys2_msgs::msg::Node>();
186 node->node_type = plansys2_msgs::msg::Node::EXPRESSION;
187 node->expression_type = getExprType(op);
188 node->node_id = tree.nodes.size();
189 tree.nodes.push_back(*node);
190
191 uint32_t left_id = 0;
192 if (n == 1) {
193 plansys2_msgs::msg::Node zero;
194 zero.node_type = plansys2_msgs::msg::Node::NUMBER;
195 zero.node_id = tree.nodes.size();
196 zero.value = 0.0;
197 tree.nodes.push_back(zero);
198 left_id = zero.node_id;
199 } else if (n == 2) {
200 left_id = args[0]->getTree(tree, d, replace)->node_id;
201 } else {
202 left_id = binaryTree(tree, d, replace, n - 1)->node_id;
203 }
204 tree.nodes[node->node_id].children.push_back(left_id);
205
206 auto right_child = args[n - 1]->getTree(tree, d, replace);
207 tree.nodes[node->node_id].children.push_back(right_child->node_id);
208
209 return node;
210 }
