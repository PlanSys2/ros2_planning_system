99 virtual void parse(const std::string & s)
100 {
101 Stringreader f(s);
102 name = f.parseName("domain");
103
104 if (DOMAIN_DEBUG) {std::cout << name << "\n";}
105
106 for (; f.getChar() != ')'; f.next()) {
107 f.assert_token("(");
108 f.assert_token(":");

113 if (!parseBlock(t, f)) {
114 f.tokenExit(t);

120 virtual bool parseBlock(const std::string & t, Stringreader & f)
121 {
122 if (t == "requirements") {
123 parseRequirements(f);
124 } else if (t == "types") {
125 parseTypes(f);
126 } else if (t == "constants") {
127 parseConstants(f);
128 } else if (t == "predicates") {
129 parsePredicates(f);
130 } else if (t == "functions") {
131 parseFunctions(f);
132 } else if (t == "action") {
133 parseAction(f);
134 } else if (t == "durative-action") {
135 parseDurativeAction(f);
136 } else if (t == "derived") {
137 parseDerived(f);
138 } else {
139 return false; // Unknown block type
140 }
141 // else if ( t == "axiom" ) parseAxiom( f );
142 return true;
143 }

162 virtual bool parseRequirement(const std::string & s)
163 {
164 if (s == "strips") {
165 strips = true;
166 } else if (s == "adl") {
167 adl = true;
168 } else if (s == "negative-preconditions") {
169 neg = true;
170 } else if (s == "conditional-effects") {
171 condeffects = true;
172 } else if (s == "typing") {
173 typed = true;
174 } else if (s == "action-cost") {
175 costs = true;
176 } else if (s == "equality") {
177 equality = true;
178 } else if (s == "durative-actions") {
179 temp = true;
180 } else if (s == "non-deterministic") {
181 nondet = true;
182 } else if (s == "universal-preconditions") {
183 universal = true;
184 } else if (s == "fluents") {
185 fluents = true;
186 } else if (s == "disjunctive-preconditions") {
187 disj = true;
188 } else if (s == "derived-predicates") {
189 derivedpred = true;
190 } else if (s == "existential-preconditions") {
191 existentialcond = true;
192 } else {
193 return false; // Unknown requirement
194 }
195 return true;
196 }

198 // get the type corresponding to a string
199 Type * getType(std::string s)
200 {
201 int i = types.index(s);
202 if (i < 0) {
203 if (s[0] == '(') {
204 i = types.insert(new EitherType(s));
205 for (unsigned k = 9; s[k] != ')'; ) {
206 unsigned e = s.find(' ', k);
207 types[i]->subtypes.push_back(getType(s.substr(k, e - k)));
208 k = e + 1;
209 }
210 } else {
211 i = types.insert(new Type(s));
212 }
213 }
214 return types[i];
215 }

225 void parseTypes(Stringreader & f)
226 {
227 if (!typed) {
228 std::cout << "Requirement :typing needed to define types\n";
229 exit(1);
230 }
231
232 // if this makes it in, probably need to define new subclass of Type
233 // if ( costs ) insert( new Type( "NUMBER" ), tmap, types );
234
235 // Parse the typed list
236 TokenStruct<std::string> ts = f.parseTypedList(false);
237
238 // bit of a hack to avoid object being the supertype
239 if (ts.index("object") >= 0) {
240 types[0]->name = "supertype";
241 types.tokenMap.clear();
242 types.tokenMap["supertype"] = 0;
243 }
244
245 // Relate subtypes and supertypes
246 for (unsigned i = 0; i < ts.size(); ++i) {
247 if (std::find(types.types.begin(), types.types.end(), ts.types[i]) == types.types.end()) {
248 if (ts.types[i].size()) {
249 getType(ts.types[i])->insertSubtype(getType(ts[i]));
250 } else {
251 getType(ts[i]);
252 }
253 }
254 }
255
256 // By default, the supertype of a type is "object"
257 for (unsigned i = 1; i < types.size(); ++i) {
258 if (types[i]->supertype == 0) {types[0]->insertSubtype(types[i]);}}
259
260 for (unsigned i = 0; DOMAIN_DEBUG && i < types.size(); ++i) {std::cout << " " << types[i];}
261 }

263 void parseConstants(Stringreader & f)
264 {
265 if (typed && !types.size()) {
266 std::cout << "Types needed before defining constants\n";
267 exit(1);
268 }
269
270 cons = true;
271
272 TokenStruct<std::string> ts = f.parseTypedList(true, types);
273
274 for (unsigned i = 0; i < ts.size(); ++i) {
275 Type * t = getType(ts.types[i]);
276 t->constants.insert(ts[i]);
277 }
278 for (unsigned i = 0; DOMAIN_DEBUG && i < types.size(); ++i) {
279 std::cout << " ";
280 if (typed) {std::cout << " " << types[i] << ":";}
281 for (unsigned j = 0; j < types[i]->constants.size(); ++j) {
282 std::cout << " " << types[i]->constants[j];
283 }
284 std::cout << "\n";
285 }

697 virtual Condition * createCondition(Stringreader & f)
698 {
699 std::string s = f.getToken();
700
701 if (s == "and") {return new And;}
702 if (s == "exists") {return new Exists;}
703 if (s == "forall") {return new Forall;}
704 if (s == "imply") {return new Imply;}
705 if (s == "assign") {return new Assign;}
706 if (s == "increase") {return new Increase;}
707 if (s == "decrease") {return new Decrease;}
708 if (s == "not") {return new Not;}
709 if (s == "oneof") {return new Oneof;}
710 if (s == "or") {return new Or;}
711 if (s == "when") {return new When;}
712 if (s == "=" || s == ">=" || s == ">" || s == "<=" || s == "<") {
713 return new CompositeExpression(s);
714 }
