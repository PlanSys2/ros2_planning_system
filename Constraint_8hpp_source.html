32// A PDDL3 modal constraint over the states of a plan: (always C), (within 10 C),
33// (sometime-before C1 C2), (hold-during 2 5 C), (at end C)...
34class Constraint : public Condition
35{
36public:
37 std::string op;
38 std::vector<double> times;
39 CondVec conds;
40
41 explicit Constraint(const std::string & o)
42 : op(o) {}
43
44 Constraint(const Constraint * c, Domain & d)
45 : op(c->op), times(c->times)
46 {
47 for (auto * cond : c->conds) {
48 conds.push_back(cond->copy(d));
49 }
50 }
51
52 ~Constraint()
53 {
54 for (auto * cond : conds) {
55 delete cond;
56 }
57 }
58
59 // The operators, and how many times and conditions each one takes
60 static bool isOperator(const std::string & o);
61 static std::pair<unsigned, unsigned> arity(const std::string & o);
62
63 void print(std::ostream & s) const
64 {
65 s << "Constraint " << op << ":\n";
66 for (auto * cond : conds) {
67 cond->print(s);
68 }
69 }
70
71 void PDDLPrint(
72 std::ostream & s, unsigned indent, const TokenStruct<std::string> & ts,
73 const Domain & d) const override;
74
75 plansys2_msgs::msg::Node::SharedPtr getTree(
76 plansys2_msgs::msg::Tree & tree, const Domain & d,
77 const std::vector<std::string> & replace = {}) const override;
78
79 void parse(Stringreader & f, TokenStruct<std::string> & ts, Domain & d);
80
81 void addParams(int m, unsigned n)
82 {
83 for (auto * cond : conds) {
84 cond->addParams(m, n);
85 }
86 }
87
88 Condition * copy(Domain & d) {return new Constraint(this, d);}
89};
